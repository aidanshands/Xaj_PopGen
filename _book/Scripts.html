#!/bin/bash

# Initialize the output file with the header
echo -e "Sample\tAssembly\tSubmitted Organism\tPredicted Organism\tSubmitted Type\tStatus\tConfidence\tBest Hit" > ani_summary_table.tsv

# Loop over each subdirectory
for dir in */; do
    # Check if the directory contains the ani-tax-report.txt file
    if [[ -f "${dir}ani-tax-report.txt" ]]; then
        # Extract the relevant lines from the ani-tax-report.txt file
        assembly=$(sed -n '1p' "${dir}ani-tax-report.txt" | cut -d':' -f2 | xargs)
        submitted_organism=$(sed -n '2p' "${dir}ani-tax-report.txt" | cut -d':' -f2 | xargs)
        predicted_organism=$(sed -n '3p' "${dir}ani-tax-report.txt" | cut -d':' -f2 | xargs)
        submitted_type=$(sed -n '4p' "${dir}ani-tax-report.txt" | cut -d':' -f2 | xargs)
        status=$(sed -n '5p' "${dir}ani-tax-report.txt" | cut -d':' -f2 | xargs)
        confidence=$(sed -n '6p' "${dir}ani-tax-report.txt" | cut -d':' -f2 | xargs)

        # Capture the ANI value from the first data row under the dashed line following the "ANI" header
        best_hit=$(awk '/^-------/{getline; print $1; exit}' "${dir}ani-tax-report.txt" | xargs)
        
        # Get the directory name without the trailing slash (which is the sample name)
        sample=${dir%/}
        
        # Append the data to the summary table
        echo -e "${sample}\t${assembly}\t${submitted_organism}\t${predicted_organism}\t${submitted_type}\t${status}\t${confidence}\t${best_hit}" >> ani_summary_table.tsv
    fi
done

echo "ANI summary table has been created in ani_summary_table.tsv"

5.2.4 Summarize_Genbank.py

#!/usr/bin/env python
# coding: utf-8

import os
import re
import pandas as pd

def extract_genome_annotation_data(file_path):
    try:
        with open(file_path, 'r', encoding='utf-8', errors='ignore') as file:
            content = file.read()
    except UnicodeDecodeError:
        print(f"Failed to decode file: {file_path}")
        return None
    
    # Find the section starting with "#Genome-Annotation-Data-START##"
    start_index = content.find("#Genome-Annotation-Data-START##")
    if start_index == -1:
        print(f"Section not found in file: {file_path}")
        return None  # Section not found
    
    # Extract the section
    section = content[start_index:]
    
    # Define the keys to extract
    keys = [
        "Genes (total)", "CDSs (total)", "Genes (coding)", 
        "CDSs (with protein)", "Genes (RNA)", "rRNAs", 
        "complete rRNAs", "partial rRNAs", "tRNAs", 
        "ncRNAs", "Pseudo Genes (total)", "CDSs (without protein)"
    ]
    
    data = {}
    for key in keys:
        match = re.search(rf"{re.escape(key)}\s*::\s*(.+)", section)
        if match:
            value = match.group(1).strip()
            # Convert values to int if possible
            try:
                value = int(value.replace(',', ''))
            except ValueError:
                pass
            data[key] = value
        else:
            data[key] = None  # Key not found or no value
            print(f"Key '{key}' not found in section of file: {file_path}")
    
    return data

def process_genbank_files(base_directory):
    result = []
    # Iterate over each subdirectory
    for root, dirs, files in os.walk(base_directory):
        for filename in files:
            if filename.endswith(".gbk"):
                sample_name = os.path.basename(root)  # Get the subdirectory name as sample name
                file_path = os.path.join(root, filename)
                data = extract_genome_annotation_data(file_path)
                if data:
                    # Add the sample name to the data
                    data['Sample'] = sample_name
                    result.append(data)
                else:
                    print(f"No data extracted from file: {file_path}")
    
    return result

def main():
    base_directory = "/Volumes/AS_Postdoc/Postdoc2024/Assembly_Workflow_PlatanusB/Step4_Annotate/Step2_Annotate/Results"  # Replace with the path to your subdirectories
    data = process_genbank_files(base_directory)
    
    if data:
        # Convert the list of dictionaries to a pandas DataFrame
        df = pd.DataFrame(data)
        
        # Reorder columns to place 'Sample' at the beginning
        cols = ['Sample'] + [col for col in df.columns if col != 'Sample']
        df = df[cols]
        
        # Save the DataFrame to a CSV file
        df.to_csv('Genome_Annotation_Summary.csv', index=False)
        
        print("Genome annotation data has been extracted and saved to Genome_Annotation_Summary.csv")
    else:
        print("No genome annotation data found in the provided files.")

if __name__ == "__main__":
    main()

# Loop through each subdirectory in the root directory
for subdir in os.listdir(root_dir):
    subdir_path = os.path.join(root_dir, subdir)
    
    # Check if it's a directory
    if os.path.isdir(subdir_path):
        sample_file_path = os.path.join(subdir_path, f'{subdir}.txt')
        
        # Check if the Sample.txt file exists in the subdirectory
        if os.path.isfile(sample_file_path):
            # Read the tab-delimited file without headers
            df = pd.read_csv(sample_file_path, sep='\t', header=None, names=columns)
            
            # Apply the first filter: pident > 80
            filter1 = df['pident'] > 80
            
            # Apply the second filter: (length/qlen) * 100 > 80
            filter2 = (df['length'] / df['qlen']) * 100 > 80
            
            # Apply both filters
            filtered_df = df[filter1 & filter2]
            
            # Define the output file path
            output_file_path = os.path.join(subdir_path, f'{subdir}.tBLASTn.filtered.txt')
            
            # Write the filtered data to a new file with headers
            filtered_df.to_csv(output_file_path, sep='\t', index=False, header=True)
